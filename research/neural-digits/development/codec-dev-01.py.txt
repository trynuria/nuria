# Targets are clipped normalized compression saving, not best-arm labels.
class Ridge:
    def __init__(self,dim,alpha=.025):
        self.a=np.repeat(np.eye(dim)[None,:,:],4,axis=0); self.b=np.zeros((4,dim));self.alpha=alpha
    def predict(self,x):
        inverses=np.linalg.solve(self.a,np.repeat(x[None,:,None],4,axis=0))[:,:,0]
        return np.sum(inverses*self.b,axis=1)+self.alpha*np.sqrt(np.maximum(0,np.sum(x*inverses,axis=1)))
    def learn(self,x,arm,reward):
        self.a[arm]+=np.outer(x,x);self.b[arm]+=reward*x

# Encoded normalized inputs are current injection; 12 exact LIF microticks.
def snn(x):
    v=np.zeros(48); counts=np.zeros(48); current=1.1*(M@x)+.65+bias
    for tick in range(12):
        v=.8*v+current; spikes=v>=1.;counts+=spikes;v-=spikes
    return counts/12
R=rng.normal(size=(48,d))*2.5; phase=rng.uniform(-np.pi,np.pi,48)
def phi(name,x):
    x=np.asarray(x)
    if name=='raw': return np.r_[1.,x]
    if name=='quadratic': return np.r_[1.,x,np.array([x[i]*x[j] for i in range(d) for j in range(i,d)])]
    if name=='rff': return np.r_[1.,x,np.cos(R@x+phase)]
    if name in ('spiking','frozen','shuffled'): return np.r_[1.,x,snn(x)]
    raise ValueError(name)
