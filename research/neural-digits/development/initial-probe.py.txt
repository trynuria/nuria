class Net:
 def __init__(self,seed,width,ticks=16,kind='snn',frozen_input=False):
  r=np.random.default_rng(seed);self.W=r.normal(0,np.sqrt(2/64),(64,width));self.H=np.full(width,.15);self.V=r.normal(0,.1,(width,10));self.C=np.zeros(10);self.ticks=ticks;self.kind=kind;self.frozen_input=frozen_input;self.t=0
  self.mm={k:np.zeros_like(getattr(self,k)) for k in ('W','H','V','C')};self.vv={k:v.copy() for k,v in self.mm.items()}
 def encode(self,x):
  current=x@self.W+self.H
  if self.kind=='continuous':
   h=np.clip((current-.1)/.8,0,1);der=((current>.1)&(current<.9))/ .8;return h,der,0
  v=np.zeros_like(current);z=np.zeros_like(current);der=np.zeros_like(current);e=np.zeros_like(current)
  for tick in range(self.ticks):
   v=.8*v+current;e=.8*e+1;sp=(v>=1).astype(float);z+=sp;der+=np.minimum(1.,1/(1+4*np.abs(v-1))**2)*e;v-=sp
  # Explicit clipped straight-through surrogate of the exact hard-spike path.
  return z/self.ticks,np.clip(der/self.ticks,0,1.5),int(z.sum())
 def forward(self,x):
  h,der,sp=self.encode(x);logits=h@self.V+self.C;return logits,h,der,sp
 def train(self,x,y,lr,l2):
  logits,h,der,_=self.forward(x);p=np.exp(logits-logits.max(1,keepdims=True));p/=p.sum(1,keepdims=True);p[np.arange(len(y)),y]-=1;p/=len(y)
  grads={'V':h.T@p+l2*self.V,'C':p.sum(0)}
  if not self.frozen_input:
   dh=(p@self.V.T)*der;grads['W']=x.T@dh+l2*self.W;grads['H']=dh.sum(0)
  self.t+=1
  for k,g in grads.items():
   g=np.clip(g,-1,1);self.mm[k]=.9*self.mm[k]+.1*g;self.vv[k]=.999*self.vv[k]+.001*g*g
   setattr(self,k,getattr(self,k)-lr*(self.mm[k]/(1-.9**self.t))/(np.sqrt(self.vv[k]/(1-.999**self.t))+1e-8))
 def state(self):return {k:getattr(self,k).copy() for k in ('W','H','V','C')}
 def restore(self,state):
  for k,v in state.items():setattr(self,k,v.copy())

  q=g.copy();alpha=[]

   al=rho*s@q;alpha.append(al);q-=al*y

  w,f,g=nw,nf,ng
