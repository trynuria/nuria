def _values(values, targets):
    if (
        not isinstance(values, (list, tuple))
        or not isinstance(targets, (list, tuple))
        or not 1 <= len(values) <= 4
        or len(values) != len(targets)
    ):
        raise ValueError("One to four paired disclosed string vectors are required")
    if any(
        type(v) is not str or len(v.encode()) > 4096
        for v in list(values) + list(targets)
    ):
        raise ValueError("Observations require bounded disclosed strings")
    return tuple(values), tuple(targets)


def relation_extension(values, targets):
    values, targets = _values(values, targets)
    result = []
    vectors = [
        (v.casefold(), y.casefold()) for v, y in zip(values, targets, strict=True)
    ]
    checks = (
        lambda v, y: v in y,
        lambda v, y: y.startswith(v),
        lambda v, y: y.endswith(v),
        lambda v, y: y in v,
        lambda v, y: v.startswith(y),
        lambda v, y: v.endswith(y),
    )
    for check in checks:
        flags = [check(v, y) for v, y in vectors]
        result.extend(
            (
                float(all(flags)),
                float(not any(flags)),
                float(any(flags) and not all(flags)),
            )
        )
    return result


def signature(node, values, targets):
    values, targets = _values(values, targets)
    base = np.asarray(_base.signature(node, values, targets), dtype=float)
    if (
        base.shape != (100,)
        or not np.isfinite(base).all()
        or np.any(base < 0)
        or np.any(base > 1)
    ):
        raise ValueError("Invalid retained operand signature")
    return base.tolist() + relation_extension(values, targets)


def normalize_family(family):
    if family in ("capitalize", "compact", "title"):
        return "other_case"
    if family not in OPS:
        raise ValueError("Unknown prospective operation family")
    return family


def with_action(observations, family):
    raw = np.asarray(observations, dtype=object)
    if raw.shape != (118,) or any(type(v) not in (int, float) for v in raw.flat):
        raise ValueError("Expected exact118-coordinate operand observation")
    vector = raw.astype(float)
    if not np.isfinite(vector).all() or np.any(vector < 0) or np.any(vector > 1):
        raise ValueError("Operand observation is not normalized")
    family = normalize_family(family)
    return vector.tolist() + [float(family == op) for op in OPS]


def action_features(node, values, targets, family):
    return with_action(signature(node, values, targets), family)


def heuristic(batch):
    """A declared strong property-based operand/family control.

    Operates on precisely the130 features supplied to learned controls. Case
    transformations are preferred for case-insensitive output constituents;
    extraction families are preferred when an operand contains a whole target.
    This is an explicit comparator, not a claimed optimal expansion policy.
    """
    x = np.asarray(batch, dtype=float)
    if (
        x.ndim != 2
        or x.shape[1] != 130
        or not np.isfinite(x).all()
        or np.any(x < 0)
        or np.any(x > 1)
    ):
        raise ValueError("Heuristic needs normalized Nx130 observations")
    rel = x[:, 36:84].reshape((-1, 16, 3))[:, :, 0]
    cf = x[:, 100:118].reshape((-1, 6, 3))[:, :, 0]
    operand_op = x[:, 84:96]
    family = x[:, 118:130]
    # Existing exact and new case-insensitive constituent evidence.
    exact_piece = rel[:, 5]
    folded_piece = cf[:, 0]
    contains_target = cf[:, 3]
    needs_case = np.maximum(0, folded_piece - exact_piece)
    # Candidate lower/upper unary properties. Future parent output is not run.
    islower = x[:, 3]
    isupper = x[:, 6]
    scores = np.repeat(
        (0.35 * folded_piece + 0.2 * contains_target + 0.15 * x[:, 99])[:, None],
        12,
        axis=1,
    )
    idx = {op: i for i, op in enumerate(OPS)}
    scores[:, idx["strip"]] += 0.15 * rel[:, 2] + 0.1 * x[:, 18]
    scores[:, idx["upper"]] += 0.7 * needs_case * islower + 0.25 * needs_case * (
        1 - isupper
    )
    scores[:, idx["lower"]] += 0.7 * needs_case * isupper + 0.25 * needs_case * (
        1 - islower
    )
    scores[:, idx["other_case"]] += 0.25 * needs_case
    for op in ("token", "split", "slice"):
        scores[:, idx[op]] += 0.45 * contains_target + 0.1 * (1 - folded_piece)
    # Avoid useless repeated case transforms as a stated observable heuristic.
    scores[:, idx["upper"]] -= 0.25 * operand_op[:, idx["upper"]]
    scores[:, idx["lower"]] -= 0.25 * operand_op[:, idx["lower"]]
    scores[:, idx["replace"]] += 0.15 * rel[:, 12]
    scores[:, idx["number"]] += 0.15 * x[:, 9]
    scores[:, idx["concat"]] += 0.5 * exact_piece
    return np.clip(np.sum(scores * family, axis=1), 0, 1)
