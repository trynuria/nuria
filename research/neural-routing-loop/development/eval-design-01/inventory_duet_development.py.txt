class Str(str):
    pass


def tokens(text):
    out=[];i=0
    while i<len(text):
        c=text[i]
        if c.isspace():i+=1;continue
        if c==';':
            while i<len(text) and text[i]!='\n':i+=1
            continue
        if c in '()':out.append(c);i+=1;continue
        if c=='"':
            i+=1;s=''
            while i<len(text):
                if text[i]=='"':
                    if i+1<len(text) and text[i+1]=='"':s+='"';i+=2;continue
                    i+=1;break
                s+=text[i];i+=1
            else:raise ValueError('unterminated string')
            out.append(Str(s));continue
        start=i
        while i<len(text) and not text[i].isspace() and text[i] not in '();':i+=1
        out.append(text[start:i])
    return out


def parse(text):
    stream=tokens(text);index=0
    def one(depth=0):
        nonlocal index
        if depth>64 or index>=len(stream):raise ValueError('sexpression bounds')
        node=stream[index];index+=1
        if type(node) is str and node=='(':
            result=[]
            while index<len(stream) and not(type(stream[index]) is str and stream[index]==')'):
                result.append(one(depth+1))
            if index>=len(stream):raise ValueError('unclosed list')
            index+=1;return result
        if type(node) is str and node==')':raise ValueError('unmatched close')
        return node
    result=[]
    while index<len(stream):result.append(one())
    return result


def ev(node,env,steps=None):
    if steps is None:steps=[0]
    steps[0]+=1
    if steps[0]>4096:raise ValueError('oracle evaluation bound')
    if isinstance(node,Str):return str(node)
    if type(node) is str:
        if node in env:return env[node]
        if node=='true':return True
        if node=='false':return False
        if re.fullmatch('-?[0-9]+',node):return int(node)
        raise ValueError('unknown oracle symbol '+node)
    op=node[0]
    if op=='ite':return ev(node[2] if ev(node[1],env,steps) else node[3],env,steps)
    if op=='let':
        values={p[0]:ev(p[1],env,steps) for p in node[1]}
        return ev(node[2],{**env,**values},steps)
    args=[ev(v,env,steps) for v in node[1:]]
    if op=='str.++':return ''.join(args)
    if op=='str.len':return len(args[0])
    if op=='str.substr':return '' if args[1]<0 or args[2]<0 else args[0][args[1]:args[1]+args[2]]
    if op=='str.at':return args[0][args[1]] if 0<=args[1]<len(args[0]) else ''
    if op=='str.indexof':return args[0].find(args[1],args[2]) if 0<=args[2]<=len(args[0]) else -1
    if op=='str.replace':return args[0].replace(args[1],args[2],1)
    if op=='str.contains':return args[1] in args[0]
    if op=='str.prefixof':return args[1].startswith(args[0])
    if op=='str.suffixof':return args[1].endswith(args[0])
    if op in ('str.to.int','str.to_int'):return int(args[0]) if re.fullmatch('[0-9]+',args[0]) else -1
    if op in ('int.to.str','str.from_int'):return str(args[0]) if args[0]>=0 else ''
    if op=='+':return sum(args)
    if op=='-':return -args[0] if len(args)==1 else args[0]-args[1]
    if op=='*':return args[0]*args[1]
    if op=='=':return all(v==args[0] for v in args[1:])
    if op=='<':return args[0]<args[1]
    if op=='<=':return args[0]<=args[1]
    if op=='>':return args[0]>args[1]
    if op=='>=':return args[0]>=args[1]
    if op=='and':return all(args)
    if op=='or':return any(args)
    if op=='not':return not args[0]
    raise ValueError('unknown oracle operator '+str(op))


def canonical(v):return json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=False)
def h(v):return hashlib.sha256(canonical(v).encode()).hexdigest()
def fingerprint(pairs):return h(sorted(set(canonical(p) for p in pairs)))
