def edit_distance(a, b):
    """Exact Levenshtein distance via bounded integer bit vectors."""
    if not a:
        return len(b)
    masks = {}
    for i, c in enumerate(a):
        masks[c] = masks.get(c, 0) | (1 << i)
    positive, negative, distance = (1 << len(a)) - 1, 0, len(a)
    high = 1 << (len(a) - 1)
    for c in b:
        equal = masks.get(c, 0)
        x = equal | negative
        horizontal = (((equal & positive) + positive) ^ positive) | equal
        ph = negative | ~(horizontal | positive)
        mh = positive & horizontal
        distance += int(bool(ph & high)) - int(bool(mh & high))
        ph = (ph << 1) | 1
        mh <<= 1
        positive = mh | ~(x | ph)
        negative = ph & x
    return distance


def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(",", ":")).encode()).hexdigest()


def depth(node):
    return 1 + max([depth(p) for p in node.get("parts", [])] +
                   ([depth(node["arg"])] if "arg" in node else []) + [0])

@dataclass
class Record:
    node: dict
    values: tuple
    size: int
    features: np.ndarray
    cursor: int = 0
    sequence: int = 0
    active: bool = True


def family_operations(arg, values, targets, family):
    """Disjoint parameter streams; every controller has the same finite grammar."""
    if family in ("strip", "upper", "lower"):
        yield {"op": family, "arg": arg}
    elif family == "other_case":
        yield from ({"op": case, "arg": arg} for case in sorted(CASES - {"strip", "upper", "lower"}))
    elif family == "token":
        for position in POSITIONS:
            for pattern in sorted(PATTERNS):
                yield {"op": "token", "arg": arg, "pattern": pattern, "index": position}
    elif family == "split":
        separators = set(" ,-_/.@:;=|\\\t()[]")
        separators.update(c for value in values for c in value if not c.isalnum())
        separators = sorted(separators)[:32]
        separators.sort(key=lambda s: not any(s in v for v in values))
        for position in POSITIONS:
            for separator in separators:
                yield {"op": "split", "arg": arg, "separator": separator, "index": position}
    elif family == "slice":
        yield from _slices(arg, values, targets)
    elif family == "replace":
        yield from ({"op": "replace", "arg": arg, "old": c, "new": ""}
                    for c in sorted(set("".join(values)))[:64])
    elif family == "number":
        yield from ({"op": "number", "arg": arg, "scale": scale, "places": places}
                    for scale in SCALES for places in (None, 0, 1, 2, 3, 4, 5, 6))
    else:
        raise ValueError("Unknown finite action family")


def search(inputs, outputs, config, scorer=None, *, budget=50000,
           max_depth=4, batch=16, collect=False, prefix_budget=10000):
    """Actual best-first expansion of executed intermediates, not preset routes.

    Search cap covers the same interpreter/candidate meter for all controllers.
    Signature/scorer CPU and counts are separately reported, never called free.
    """
    began = time.process_time()
    rows = [fields(x, config) for x in inputs]
    targets = tuple(outputs)
    if not rows or len(rows) != len(targets):
        raise ValueError("Paired nonempty disclosed examples required")
    prefix_meter = _Meter(max(1, min(prefix_budget, budget)))
    # Same strong visible-property prefix and budget for EVERY controller.
    # This executes an existing tool; it is not an unmetered oracle/extra route.
    comparisons = []
    contained = []
    for field in range(min(map(len, rows))):
        values = [row[field] for row in rows]
        contained.append(sum(y.casefold() in v.casefold() for v, y in zip(values, targets)) / len(targets))
        comparisons.extend(sum(f(v) == y for v, y in zip(values, targets)) / len(targets)
                           for f in (lambda v: v, str.strip, str.upper, str.lower))
    prefix_profile = ('slice_first' if max(contained, default=0) >= .75 and
                      max(comparisons, default=0) < .75 and
                      sum(map(len, targets)) / 256 <= .065 else 'default')
    prefix_nodes = []
    prefix_by_node = {}
    capture_sources = {str(path.name): hashlib.sha256(path.read_bytes()).hexdigest()
                       for path in (Path(__file__), Path(__file__).parent / "captured_composition.py",
                                    Path(__file__).parent / "captured_routed_text_search.py",
                                    Path("./cognition/text_skills.py"),
                                    Path("./cognition/text_composition.py"),
                                    Path("./cognition/routed_text_search.py"))}
    capture_binding = digest({"inputs": inputs, "outputs": outputs, "configuration": config,
                              "profile": prefix_profile, "mode": "deduplicated", "prefix_cap": prefix_budget, "total_cap": budget, "sources": capture_sources})
    def capture(node, values, size):
        # Called only after original attempt executed ALL disclosed rows and
        # charged its output data. Partial WorkLimit attempts cannot enter.
        item = {"node": json.loads(json.dumps(node)), "values": list(values), "size": size}
        key = digest(node)
        if key in prefix_by_node and prefix_by_node[key] != item:
            raise ValueError("A prefix AST has inconsistent observed values")
        prefix_nodes.append(item)
        prefix_by_node[key] = item

    prefix_node = None
    if prefix_budget:
        try:
            if prefix_budget >= budget:
                # The standalone current-product comparator remains untouched.
                prefix_node = infer_routed(inputs, outputs, config, prefix_meter, profile=prefix_profile)
            else:
                prefix_node = captured_infer(inputs, outputs, config, prefix_meter, profile=prefix_profile, observer=capture)
        except WorkLimit:
            pass
    if prefix_node is not None:
        return {'program': validate_program({'version': VERSION, 'expr': prefix_node}, config),
                'status': 'solved_prefix', 'work': prefix_meter.values,
                'signature_count': 0, 'signature_cpu_seconds': 0.,
                'scored_count': 0, 'scorer_cpu_seconds': 0.,
                'cpu_seconds': time.process_time() - began,
                'candidate_records': [] if collect else None,
                'prefix_profile': prefix_profile, 'prefix_work': prefix_meter.values, 'prefix_capture_binding_sha256': capture_binding, 'prefix_capture_sha256': digest(prefix_nodes), 'prefix_capture_count': len(prefix_nodes)}
    if prefix_meter.values['total'] >= budget:
        return {'program': None, 'status': 'work_limit', 'work': prefix_meter.values,
                'signature_count': 0, 'signature_cpu_seconds': 0.,
                'scored_count': 0, 'scorer_cpu_seconds': 0.,
                'cpu_seconds': time.process_time() - began,
                'candidate_records': [] if collect else None,
                'prefix_profile': prefix_profile, 'prefix_work': prefix_meter.values, 'prefix_capture_binding_sha256': capture_binding, 'prefix_capture_sha256': digest(prefix_nodes), 'prefix_capture_count': len(prefix_nodes)}
    meter = _Meter(max(1, budget - prefix_meter.values['total']))
    records, seen, frontier, evidence = [], {}, [], []
    found = None
    stopped_because = None
    sequence = 0
    feature_cpu = score_cpu = 0.
    feature_count = score_count = 0
    warm_frontier_sha256 = digest([])
    scorer = heuristic if scorer is None else scorer

    def charge_candidate():
        if meter.values["candidates"] + prefix_meter.values["candidates"] >= config["max_candidates"]:
            raise WorkLimit("combined_candidate_limit")
        meter.charge("candidates")

    def register(node, values, size):
        nonlocal feature_cpu, feature_count, found
        if values == targets:
            found = node
            return None
        node_depth = depth(node)
        prior = seen.get(values, [])
        if any(r.active and r.size <= size and depth(r.node) <= node_depth for r in prior) or not any(values):
            meter.values["semantic_duplicates"] += 1
            return None
        for r in prior:
            if r.size >= size and depth(r.node) >= node_depth:
                r.active = False
        start = time.process_time()
        features = np.asarray(signature(node, values, targets), dtype=float)
        feature_cpu += time.process_time() - start
        feature_count += 1
        record = Record(node, values, size, features, sequence=len(records))
        records.append(record)
        seen.setdefault(values, []).append(record)
        if collect:
            evidence.append({"node": node, "values": list(values), "features": features.tolist()})
        return record

    def attempt(node, inner=None):
        key = digest(node)
        if key in prefix_by_node:
            # Reusing the already fully executed vector has an explicit lookup
            # charge. Its original interpreter/serialization work stays paid
            # exactly once in prefix_meter; no inferred subnodes are imported.
            meter.charge("primitive_attempts")
            prior = prefix_by_node[key]
            return register(node, tuple(prior["values"]), prior["size"])
        charge_candidate()
        meter.data(json.dumps(node, sort_keys=True, separators=(",", ":")))
        try:
            size = shape(node, config)
            if inner is None:
                values = tuple(evaluate(node, row, meter, config) for row in rows)
            else:
                cached = {**node, "arg": {"op": "field", "index": 0}}
                values = tuple(evaluate(cached, [v], meter, config) for v in inner)
        except (ValueError, IndexError, KeyError, InvalidOperation) as error:
            if isinstance(error, WorkLimit):
                raise
            meter.values["errors"] += 1
            return None
        meter.data(json.dumps(values, separators=(",", ":")))
        return register(node, values, size)

    def enqueue_actions(actions):
        nonlocal score_cpu, score_count, sequence
        actions = [(r, f, cursor) for r, f, cursor in actions
                   if r is not None and r.active and depth(r.node) < max_depth]
        if not actions:
            return
        observed = np.asarray([np.concatenate((r.features, np.asarray([float(f == op) for op in OPS])))
                               for r, f, _ in actions])
        start = time.process_time()
        scores = np.asarray(scorer(observed), dtype=float)
        score_cpu += time.process_time() - start
        score_count += len(actions)
        if scores.shape != (len(actions),) or not np.isfinite(scores).all():
            raise ValueError("Invalid operation-edge scores")
        for (record, family, cursor), score in zip(actions, scores, strict=True):
            sequence += 1
            priority = float(score) - .10 * (record.size - 1) - .0005 * cursor
            heapq.heappush(frontier, (-priority, sequence, record, family, cursor))

    def enqueue(new):
        enqueue_actions([(r, family, 0) for r in new if r is not None for family in ACTION_FAMILIES])

    def concatenation():
        nonlocal found
        # Visible output prefixes only. Every transition and delivery is charged.
        atoms = [r for r in records if r.active and all(v in y for v, y in zip(r.values, targets))]
        queue = [(0, 0, (0,) * len(targets), ())]
        best_cost = {}
        serial = 0
        while queue:
            cost, _, positions, parts = heapq.heappop(queue)
            if parts and cost != best_cost.get((positions, len(parts))):
                continue
            if positions == tuple(map(len, targets)) and parts:
                if len(parts) == 1:
                    found = parts[0]
                else:
                    attempt({"op": "concat", "parts": list(parts)})
                return
            if len(parts) >= config["max_parts"]:
                continue
            literals = set()
            tail = targets[0][positions[0]:]
            for length in range(1, min(8, len(tail)) + 1):
                literal = tail[:length]
                if all(y.startswith(literal, p) for y, p in zip(targets, positions)):
                    literals.add(literal)
            options = [(r.node, r.values, r.size) for r in atoms]
            options.extend(({"op": "constant", "value": s}, (s,) * len(targets), 1)
                           for s in sorted(literals))
            for node, values, size in options:
                charge_candidate()
                meter.data(json.dumps(values, separators=(",", ":")))
                if cost + size + 1 > config["max_ops"]:
                    continue
                if not all(y.startswith(v, p) for y, v, p in zip(targets, values, positions)):
                    continue
                future = tuple(p + len(v) for p, v in zip(positions, values))
                key = (future, len(parts) + 1)
                if future == positions or best_cost.get(key, float('inf')) <= cost + size:
                    continue
                best_cost[key] = cost + size
                serial += 1
                heapq.heappush(queue, (cost + size, serial, future, (*parts, node)))

    try:
        for x in inputs + outputs:
            meter.data(x)
        if len(set(targets)) == 1:
            attempt({"op": "constant", "value": targets[0]})
        initial = []
        for item in prefix_nodes:
            node, values, size = item["node"], tuple(item["values"]), item["size"]
            if digest(node) not in prefix_by_node or shape(node, config) != size:
                raise ValueError("Prefix cache structural binding failed")
            if values == targets:
                raise ValueError("A completed prefix solution was not returned")
            candidate = register(node, values, size)
            if candidate is not None:
                initial.append(candidate)
        for i in range(min(map(len, rows))):
            if found is not None:
                break
            node = {"op": "field", "index": i}
            if digest(node) in prefix_by_node:
                continue
            candidate = attempt(node)
            if candidate is not None:
                initial.append(candidate)
        warm_frontier_sha256 = digest([{ "node": r.node, "values": list(r.values), "size": r.size}
                                        for r in initial if r.active])
        enqueue(initial)
        iterations = 0
        iterators = {}
        while frontier and found is None:
            _, _, record, family, cursor = heapq.heappop(frontier)
            if not record.active:
                continue
            action_key = (record.sequence, family)
            if action_key not in iterators:
                iterators[action_key] = iter(family_operations(record.node, record.values, targets, family))
            generated, exhausted = [], False
            for _ in range(batch):
                try:
                    node = next(iterators[action_key])
                except StopIteration:
                    exhausted = True
                    break
                cursor += 1
                new = attempt(node, record.values)
                if new is not None:
                    generated.append(new)
                if found is not None:
                    break
            enqueue(generated)
            if not exhausted:
                enqueue_actions([(record, family, cursor)])
            iterations += 1
            if iterations % 4 == 0 and found is None:
                concatenation()
        if found is None:
            concatenation()
        status = "solved" if found is not None else "not_found"
    except WorkLimit as error:
        status = "work_limit"
        stopped_because = str(error)
    program = validate_program({"version": VERSION, "expr": found}, config) if found is not None else None
    full_work = {k: meter.values[k] + prefix_meter.values[k] for k in meter.values}
    return {"program": program, "status": status, "stopped_because": stopped_because, "work": full_work,
            "signature_count": feature_count, "signature_cpu_seconds": feature_cpu,
            "scored_count": score_count, "scorer_cpu_seconds": score_cpu,
            "cpu_seconds": time.process_time() - began,
            "candidate_records": evidence if collect else None,
            "prefix_profile": prefix_profile, "prefix_work": prefix_meter.values, "prefix_capture_binding_sha256": capture_binding, "prefix_capture_sha256": digest(prefix_nodes), "prefix_capture_count": len(prefix_nodes), "warm_frontier_sha256": warm_frontier_sha256}


def edge_training(delivered_programs, inputs, outputs, candidates, config):
    """Observed operand/family labels from actually buyer-accepted AST edges.

    Proper children of earlier accepted ASTs are evaluated and included even
    when the common exploratory search missed them. This is checked past
    artifact feedback, supplied identically to every controller. Parent root
    edges are permitted; the parent output is never a feature. Nonmembership
    is weak negative evidence, not proof that a family is universally useless.
    """
    begin = time.process_time()
    rows = [fields(value, config) for value in inputs]
    targets = tuple(outputs)
    positive = set()
    source_programs = {}
    teacher_nodes = {}
    meter = _Meter(1000000)
    def visit(node, program_hash):
        if "arg" in node:
            family = normalize_family(node["op"])
            child = node["arg"]
            values = tuple(evaluate(child, row, meter, config) for row in rows)
            if family in ACTION_FAMILIES and values != targets:
                positive.add((values, family))
                source_programs.setdefault((values, family), program_hash)
                identity = digest(child)
                teacher_nodes[identity] = {"node": child, "values": list(values),
                                           "features": signature(child, values, targets)}
            visit(child, program_hash)
        for child in node.get("parts", []):
            visit(child, program_hash)
        if "then" in node or "else" in node:
            raise ValueError("Conditional programs are outside edge-teaching DSL")
    for program in delivered_programs:
        checked = validate_program(program, config)
        visit(checked["expr"], digest(checked))
    pool = {digest(c["node"]): c for c in candidates}
    original = set(pool)
    pool.update(teacher_nodes)
    result = []
    canonical_source = min(map(digest, delivered_programs))
    for candidate in pool.values():
        node, values = candidate["node"], tuple(candidate["values"])
        if values == targets:
            continue
        base = signature(node, values, targets)
        for family in ACTION_FAMILIES:
            features = np.concatenate((base, np.asarray([float(family == op) for op in OPS]))).tolist()
            result.append({"features": features,
                           "label": int((values, family) in positive),
                           "candidate_sha256": digest({"operand": node, "family": family}),
                           "program_sha256": source_programs.get((values, family), canonical_source)})
    return {"records": result, "teacher_candidates": list(teacher_nodes.values()),
            "teacher_operand_replay_work": meter.values,
            "teacher_replay_cpu_seconds": time.process_time() - begin,
            "positive_semantic_edges": len(positive),
            "teacher_operand_nodes": len(teacher_nodes),
            "teacher_operand_nodes_not_in_exploration": len(set(teacher_nodes) - original)}
