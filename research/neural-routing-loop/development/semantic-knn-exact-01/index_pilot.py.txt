from pathlib import Path

P=Path(__file__).parent;D=P.parent/'semantic-priority-dev-06';rng=np.random.default_rng(20261028)
count=0;fallback=0
for dim in(1,3,16,130):
    x=rng.integers(0,2,(127,dim)).astype(float);x[64:]=x[:63]
    if dim>3:x[:,-2:]=rng.random((127,2))
    y=rng.integers(0,2,127).astype(float)
    for k in(1,2,9,128):
        model=SimpleNamespace(configuration={'method':'knn','dimension':dim,'neighbors':k},training_sha256='a'*64,reference_x=x,reference_y=y,positive_count=int(y.sum()),records_count=len(y))
        optimized=IndexedExactKNN(model,capacity=16)
        for i in range(256):
            q=x[i%127].copy()if i%3 else rng.random(dim)
            distance=np.sum((x-q)**2,axis=1);indices=np.argsort(distance,kind='stable')[:min(k,len(x))]
            weights=1/(distance[indices]+.001);weights*=np.where(y[indices]==1,.5/model.positive_count,.5/(127-model.positive_count))
            expected=float(np.dot(weights,y[indices])/weights.sum());observed,found=optimized._one(q)
            if not np.array_equal(indices,found)or observed!=expected:raise ValueError('Synthetic index exactness failed')
            count+=1
        fallback+=optimized.fallback_queries
model=SemanticGuidance.from_evidence(json.loads((D/'model-0-knn.json').read_text()));optimized=IndexedExactKNN(model)
observations=[]
with(D/'collection-0.jsonl').open()as stream:
    for line in stream:
        observations.extend(r['features']for r in json.loads(line)['selected_records'])
        if len(observations)>=100:break
observations=observations[:100]
started=time.process_time();expected=model.score(observations);oldcpu=time.process_time()-started
started=time.process_time();actual=optimized.score(observations);newcpu=time.process_time()-started
if not np.array_equal(expected,actual):raise ValueError('Actual curriculum indexed score differs')

logged=json.loads((P/'actual-score-calls.json').read_text());optimized.reset();started=time.process_time()

    if optimized.score(call['features']).tolist()!=call['scores']:raise ValueError('Actual search indexed scores differ')
