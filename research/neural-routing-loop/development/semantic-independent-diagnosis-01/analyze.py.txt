def digest(x):return hashlib.sha256(json.dumps(x,sort_keys=True,separators=(',',':')).encode()).hexdigest()
def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def visit(node):
 yield node
 for p in node.get('parts',[]):yield from visit(p)
 if 'arg'in node:yield from visit(node['arg'])
 if node['op']=='if_equal':
  yield from visit(node['then']);yield from visit(node['else'])
def summarize_collisions(records,feature_key):
 groups=defaultdict(list)
 for r in records:groups[feature_key(r['features'])].append(r['label'])

 return dict(unique_feature_vectors=len(groups),conflicting_vectors=len(conflicting),records_in_conflicting_vectors=sum(map(len,conflicting)),unweighted_minimum_training_errors=sum(min(sum(v),len(v)-sum(v))for v in groups.values()),scope='Empirical training membership classification only, not a bound on actual search usefulness or unseen generalization')
