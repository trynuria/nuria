class SNN:

  if self.continuous:h=np.clip((drive-.1)/.8,0,1);d=((drive>.1)&(drive<.9))/.8;return h@self.V+self.C,h,d
  v=np.zeros_like(drive);counts=np.zeros_like(v);der=np.zeros_like(v);elig=np.zeros_like(v)
  for tick in range(16):
   v=.8*v+drive;elig=.8*elig+1;sp=v>=1;counts+=sp;der+=elig/(1+4*np.abs(v-1))**2;v-=sp
  h=counts/16
